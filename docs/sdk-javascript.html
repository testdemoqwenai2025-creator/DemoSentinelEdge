JavaScript SDK

Installation

npm install @aisensoredgecomp/sdk
# or
yarn add @aisensoredgecomp/sdk

Quick start (Node.js)

import {{ AISensorEdgeComp }} from '@aisensoredgecomp/sdk';

const client = new AISensorEdgeComp({{
  apiUrl: 'https://api.aisensoredgecomp.ai',
  email: 'you@company.com',
  password: 'your-password',
}});

// List sensors
const sensors = await client.sensors.list({{ sensorKind: 'vibration' }});
console.log(`${{sensors.length}} vibration sensors`);

// Run anomaly detection
const result = await client.ml.inferAnomaly({{
  sensorId: 'vib_a3_l3_compressor_a_bearing1',
  window: [4.0, 4.1, 4.2, 8.5, 4.0, 4.1],
  sensorKind: 'vibration',
}});
console.log(`Anomaly: ${{result.anomalyScore}}`);
