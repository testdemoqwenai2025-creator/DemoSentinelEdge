Code samples

Quick start with the API

Python (requests)

# Get a JWT token
import requests

resp = requests.post("https://api.aisensoredgecomp.ai/api/v1/auth/login", json={
    "email": "you@company.com",
    "password": "your-password",
})
token = resp.json()["token"]
headers = {"Authorization": f"Bearer {token}"}

# List sensors
sensors = requests.get(
    "https://api.aisensoredgecomp.ai/api/v1/sensors",
    headers=headers,
    params={"sensor_kind": "vibration", "page_size": 50},
).json()

# Get readings for a sensor
readings = requests.get(
    f"https://api.aisensoredgecomp.ai/api/v1/sensors/{sensors['sensors'][0]['sensor_id']}/readings",
    headers=headers,
    params={"limit": 100},
).json()

# Ask a natural-language question
answer = requests.post(
    "https://api.aisensoredgecomp.ai/api/v1/search",
    headers=headers,
    json={"query": "What is the current status of compressor A?"},
).json()
print(answer["answer"])

JavaScript (fetch)

// Sign in
const loginResp = await fetch("https://api.aisensoredgecomp.ai/api/v1/auth/login", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({
    email: "you@company.com",
    password: "your-password",
  }),
});
const { token } = await loginResp.json();

// Query sensors
const sensorsResp = await fetch("https://api.aisensoredgecomp.ai/api/v1/sensors", {
  headers: { Authorization: `Bearer ${token}` },
});
const sensors = await sensorsResp.json();

curl

# Sign in
curl -X POST https://api.aisensoredgecomp.ai/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"you@company.com","password":"your-password"}'

# List sensors (replace TOKEN)
curl https://api.aisensoredgecomp.ai/api/v1/sensors \
  -H "Authorization: Bearer TOKEN"

# Ask a question
curl -X POST https://api.aisensoredgecomp.ai/api/v1/search \
  -H "Authorization: Bearer TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"query":"What is your TAM?"}'
